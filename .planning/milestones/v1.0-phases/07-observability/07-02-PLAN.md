         if (process.env.NEXT_RUNTIME === 'nodejs') {

    3. Create `sentry.server.config.ts` at project root — per D-01, strip ALL user data. IMPORTANT: Export the beforeSend function as a named export so it can be imported and tested directly in sentry.test.ts. Do NOT define it inline-only or copy it into tests:
       ```typescript
       import * as Sentry from '@sentry/nextjs'
       import type { Event } from '@sentry/nextjs'

       export function beforeSend(event: Event): Event {
         if (event.request) {
           delete event.request.data
           delete event.request.cookies
           if (event.request.env) {
             delete event.request.env['REMOTE_ADDR']
           }
         }
         delete event.user
         return event
       }

       Sentry.init({
         dsn: process.env.NEXT_PUBLIC_SENTRY_DSN,
         tracesSampleRate: 0.1,
         beforeSend,
       })
       ```

       import NextError from 'next/error'

       export default function GlobalError({ error }: { error: Error & { digest?: string } }) {
         useEffect(() => {
           Sentry.captureException(error)
