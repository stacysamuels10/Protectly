**Search input (per D-02, D-03):**

Add imports:
```typescript
import { useRef } from 'react' // add to existing React imports
import { Search } from 'lucide-react'
import { Input } from '@/components/ui/input'
```

Add debounce ref and handler inside the component:
```typescript
const debounceRef = useRef<ReturnType<typeof setTimeout> | null>(null)
const [searchInput, setSearchInput] = useState(q) // local input state for immediate display

const handleSearchChange = useCallback((value: string) => {
  setSearchInput(value)
  if (debounceRef.current) clearTimeout(debounceRef.current)
  debounceRef.current = setTimeout(() => {
    updateParams({ q: value || null, page: null })
  }, 300)
}, [updateParams])
```

In the filter toolbar, after the `TabsList` and on the same row (right side), add:
```tsx
<div className="relative">
  <Search className="absolute left-2.5 top-2.5 h-4 w-4 text-muted-foreground" />
  <Input
    placeholder="Search by email..."
    aria-label="Search activity by email"
    value={searchInput}
    onChange={(e) => handleSearchChange(e.target.value)}
    className="pl-8 w-64"
  />
</div>
```

The filter toolbar row layout: `div.flex.items-center.justify-between.flex-wrap.gap-2` containing `TabsList` (left) and the search div (right).
